# Recommended install order on a new machine:
#
#   conda create -n unilidar python=3.8 -y
#   conda activate unilidar
#
#   # 1) PyTorch (CUDA 11.3)
#   pip install torch==1.10.1+cu113 torchvision==0.11.2+cu113 torchaudio==0.10.1+cu113 \
#       -f https://download.pytorch.org/whl/cu113/torch_stable.html

#   pip install mmcv-full==1.4.0 -f https://download.openmmlab.com/mmcv/dist/cu113/torch1.10.0/index.html
#   pip install mmdet==2.14.0 mmsegmentation==0.14.1 mmengine==0.10.3
#
#   # 3) mmdet3d (from source, v0.17.1)
#   git clone https://github.com/open-mmlab/mmdetection3d.git && cd mmdetection3d \
#       && git checkout v0.17.1 && pip install -v -e . && cd ..
#
#   # 4) CUDA-specific wheels
#   pip install spconv-cu113==2.3.6
#   pip install torch-scatter==2.0.9 \
#       -f https://data.pyg.org/whl/torch-1.10.0+cu113.html
#
#   # 5) Remaining dependencies
#   pip install -r requirements.txt
#
#   # 6) LiSeer plugin
#   cd LiSeer-Simple && export PYTHONPATH=. && pip install -v -e .
#
# Notes:
# - torch-scatter in this env was installed from a local wheel; use PyG index above.
